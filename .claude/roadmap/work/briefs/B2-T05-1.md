// buildingContext.ts
export type SummaryAfterSaveResult =
  | { state: 'no-building' }                       // store holds no building file
  | { state: 'no-match' }                          // no sheet matches by id or path (weak matches are ignored)
  | { state: 'written'; projectUuid: string; patchedSheets: number };

export async function summaryAfterSave(args: {
  store: BuildingStore;
  project: ProjectDocument;                        // from scene.exportProject(); project.projectUuid undefined -> { state: 'no-match' }
  document: { fileName: string; path?: string; sizeBytes: number; modifiedMs: number };
  pageInfos: readonly { widthPt: number; heightPt: number; rotationDegrees: number }[];
  stampTypeName: (id: string) => string | null;
  stampImageName: (id: string) => string | null;
  now?: () => Date;                                // default () => new Date()
}): Promise<SummaryAfterSaveResult>;

/** The two name lookups of the quantity tables, from a list of stamp definitions. Same rules as handleExportQuantities. */
export function stampNameLookups(definitions: readonly StampDefinition[]): { stampTypeName: (id: string) => string | null; stampImageName: (id: string) => string | null };

Steps of summaryAfterSave:
1. `store.read()`; undefined -> no-building. `parseBuildingFile(JSON.parse(json))` (errors propagate).
2. `findSheets(file, { projectUuid, path, fileName, sizeBytes })`, keep only strength 'id' or 'path'. None -> no-match.
3. `buildSheetSummary({... savedAt: now().toISOString(), sourceSizeBytes: document.sizeBytes, sourceModifiedMs: document.modifiedMs })`; `store.writeSummary(projectUuid, JSON.stringify(summary))`. Write the summary BEFORE the building file patch.
4. Sheets among the matches with `projectUuid === null` get `withProjectUuid`. If any: `store.write(serializeBuildingFile(patched), revision)`. On `StoreConflictError` re-read, re-parse, re-apply the patch by sheet id (sheets that are gone are skipped) and retry ONCE; a second conflict propagates. Return `patchedSheets` = number of sheets patched (0 if none needed).

stampNameLookups copies this logic from handleExportQuantities: typeName = `getStampDefinition(id, definitions)?.label ?? null`; imageName = iconRef file name (last path segment) unless iconRef starts with data: or blob:, else null. Use the same `getStampDefinition` import that App.tsx uses (find where it comes from).
