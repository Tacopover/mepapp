```ts
export const SHEET_SUMMARY_VERSION = 1;
export interface SheetSummaryPage { calibrated: boolean; pageUnitsPerRealUnit: number | null; widthPt: number; heightPt: number; rotationDegrees: number }
export interface SheetSummary {
  version: number;
  projectUuid: string;
  savedAt: string;            // ISO time
  sourceSizeBytes: number;
  sourceModifiedMs: number;
  /** key = page index as text. Width and height are the DISPLAYED size after the page's own /Rotate (PageInfo of pdf-engine is already post-rotation). */
  pages: Record<string, SheetSummaryPage>;
  tables: QuantityTable[];
  warnings: string[];
}
export class SheetSummaryError extends Error { constructor(message: string, readonly issues: ValidationIssue[]) }
export function buildSheetSummary(args: {
  project: ProjectDocument;
  savedAt: string;
  sourceSizeBytes: number;
  sourceModifiedMs: number;
  /** Page sizes as displayed (post-rotation), index = 0-based page. Same shape as PageInfo of @mepapp/pdf-engine. core must not import that package: declare the shape inline. */
  pageInfos: readonly { widthPt: number; heightPt: number; rotationDegrees: number }[];
  stampTypeName: (id: string) => string | null;
  stampImageName: (id: string) => string | null;
}): SheetSummary;
export function parseSheetSummary(raw: unknown): SheetSummary;   // throws SheetSummaryError
```
